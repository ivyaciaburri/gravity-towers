function move(x1,y1,x2,y2){ const tower = board[y1][x1]; const target = board[y2][x2]; const fromCell = boardDiv.children[y1 * WIDTH + x1]; const toCell = boardDiv.children[y2 * WIDTH + x2]; const pieceDiv = fromCell.querySelector(".piece"); if(!pieceDiv) return; // Remove moving piece from board immediately board[y1][x1] = null; // Simple 1-on-1 capture, no animation if(tower.height ===1 && target && target.height ===1){ tower.height += target.height; board[y2][x2] = tower; render(); // instant update applyGravityAnimated(x2,y2); // in case tower grows to >1 return; } // Otherwise: tower or multi-piece gravity animation const rect = fromCell.getBoundingClientRect(); pieceDiv.style.position = "fixed"; pieceDiv.style.left = rect.left + "px"; pieceDiv.style.top = rect.top + "px"; pieceDiv.style.zIndex = 1000; document.body.appendChild(pieceDiv); const destRect = toCell.getBoundingClientRect(); const dx = Math.abs(destRect.left - rect.left); const dy = Math.abs(destRect.top - rect.top); const distance = Math.sqrt(dx*dx + dy*dy); const duration = 0.8 + distance/200; requestAnimationFrame(()=>{ pieceDiv.style.transition = `all ${duration}s cubic-bezier(0.4,0,0.2,1)`; pieceDiv.style.left = destRect.left + "px"; pieceDiv.style.top = destRect.top + "px"; }); pieceDiv.addEventListener("transitionend", ()=>{ if(target) tower.height += target.height; board[y2][x2] = tower; pieceDiv.style.position = "relative"; pieceDiv.style.left = "0px"; pieceDiv.style.top = "0px"; pieceDiv.style.zIndex = 1; pieceDiv.textContent = tower.height; toCell.appendChild(pieceDiv); applyGravityAnimated(x2,y2); }, {once:true}); } function switchTurn(){ currentPlayer = currentPlayer==="red"?"blue":"red"; // Skip turn if no pieces left if(!hasPieces(currentPlayer)){ statusDiv.textContent = currentPlayer.charAt(0).toUpperCase()+currentPlayer.slice(1) + " has no pieces, skipping turn."; currentPlayer = currentPlayer==="red"?"blue":"red"; } statusDiv.textContent = currentPlayer.charAt(0).toUpperCase()+currentPlayer.slice(1)+"'s Turn"; } function hasPieces(color){ for(let row of board) for(let cell of row) if(cell && cell.color===color) return true; return false; }
